name: tests

on:
  push:
    branches: [main]
  pull_request:

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: 의존성 설치
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
      - name: 단위·시나리오 테스트 (Liner 키 없이)
        env:
          LINER_OFFLINE: "1"
        run: python -m unittest discover -s tests -p "test_*.py" -v
      - name: 웹 UI·API E2E (HTTP)
        env:
          LINER_OFFLINE: "1"
        run: python -m unittest tests.e2e_http -v

# ─────────────────────────────────────────────────────────────
# 사용법: 이 파일을 `.github/workflows/tests.yml` 로 복사해 커밋하면 GitHub Actions가
# 키 없이(오프라인 모드) 단위·E2E 테스트를 실행합니다.
# (피클 GitHub 연결은 workflow 파일 쓰기 권한이 없어 저장소에는 .txt 로 둡니다.)
# ─────────────────────────────────────────────────────────────
